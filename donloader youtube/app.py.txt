import streamlit as st
import yt_dlp
import os

st.title("📥 موقع تحميل فيديوهات يوتيوب")
st.write("أدخل رابط أي فيديو من يوتيوب أدناه لتحميله بسهولة وبأفضل جودة.")

# خانة كتابة الرابط
url = st.text_input("أدخل رابط يوتيوب هنا:")

if st.button("تحميل الفيديو"):
    if url:
        with st.spinner("جاري معالجة وتحميل الفيديو، يرجى الانتظار..."):
            try:
                ydl_opts = {
                    'format': 'best',
                    'outtmpl': 'downloads/%(title)s.%(ext)s',
                    'extractor_args': {'youtube': {'player_client': ['android', 'web']}}
                }
                
                with yt_dlp.YoutubeDL(ydl_opts) as ydl:
                    info_dict = ydl.extract_info(url, download=True)
                    filename = ydl.prepare_filename(info_dict)
                
                st.success("تم التحميل بنجاح!")
                
                # زر لتنزيل الملف مباشرة للمستخدم
                with open(filename, "rb") as file:
                    st.download_button(
                        label="اضغط هنا لتنزيل الفيديو على جهازك",
                        data=file,
                        file_name=os.path.basename(filename),
                        mime="video/mp4"
                    )
            except Exception as e:
                st.error(f"حدث خطأ: {e}")
    else:
        st.warning("الرجاء إدخال رابط صحيح أولاً.")
